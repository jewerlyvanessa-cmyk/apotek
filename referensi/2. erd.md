# ERD PostgreSQL Lengkap

# 1. tenants

```sql
CREATE TABLE tenants (
    id UUID PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    phone VARCHAR(30),
    email VARCHAR(255),
    address TEXT,
    subscription_plan VARCHAR(50),
    subscription_expired_at TIMESTAMP,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);
```

---

# 2. branches

```sql
CREATE TABLE branches (
    id UUID PRIMARY KEY,
    tenant_id UUID REFERENCES tenants(id),
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50),
    address TEXT,
    phone VARCHAR(30),
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);
```

---

# 3. users

```sql
CREATE TABLE users (
    id UUID PRIMARY KEY,
    tenant_id UUID REFERENCES tenants(id),
    branch_id UUID REFERENCES branches(id),
    full_name VARCHAR(255),
    email VARCHAR(255) UNIQUE,
    phone VARCHAR(30),
    password_hash TEXT,
    role VARCHAR(50),
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);
```

# 6. medicines

```sql
CREATE TABLE medicines (
    id UUID PRIMARY KEY,
    tenant_id UUID REFERENCES tenants(id),
    category_id UUID REFERENCES medicine_categories(id),
    supplier_id UUID REFERENCES suppliers(id),
    sku VARCHAR(100),
    barcode VARCHAR(100),
    name VARCHAR(255),
    unit VARCHAR(50),
    buy_price NUMERIC(18,2),
    sell_price NUMERIC(18,2),
    min_stock INTEGER DEFAULT 0,
    requires_prescription BOOLEAN DEFAULT false,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);
```

# 7. medicine_batches

```sql
CREATE TABLE medicine_batches (

    medicine_id UUID REFERENCES medicines(id),
    batch_number VARCHAR(100),
    expired_date DATE,
    buy_price NUMERIC(18,2),
    sell_price NUMERIC(18,2),
    created_at TIMESTAMP DEFAULT NOW()
);
```

---

# 8. stocks

```sql
CREATE TABLE stocks (
    id UUID PRIMARY KEY,
    tenant_id UUID REFERENCES tenants(id),
    branch_id UUID REFERENCES branches(id),
    medicine_id UUID REFERENCES medicines(id),
    batch_id UUID REFERENCES medicine_batches(id),
    quantity INTEGER DEFAULT 0,
    reserved_quantity INTEGER DEFAULT 0,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(branch_id, medicine_id, batch_id)
);
```

---

# 9. stock_movements

```sql
CREATE TABLE stock_movements (

    batch_id UUID REFERENCES medicine_batches(id),

    reference_type VARCHAR(50),

    notes TEXT,

```

# 10. orders

```sql
    paid_at TIMESTAMP,

```

# Relasi Utama ERD

## Tenant
- tenant punya banyak cabang
- tenant punya banyak user
- tenant punya banyak obat

## Branch
- branch punya stok sendiri
- branch punya order sendiri
- branch punya kasir sendiri

## Orders
- order punya banyak item
- order punya pembayaran

## Medicines
- medicine punya batch
- medicine punya stok
- medicine punya histori mutasi

# Index Penting PostgreSQL

```sql
CREATE INDEX idx_stocks_branch_medicine
ON stocks(branch_id, medicine_id);

CREATE INDEX idx_orders_branch_status
ON orders(branch_id, status);

CREATE INDEX idx_order_items_order
ON order_items(order_id);

CREATE INDEX idx_stock_movements_created
ON stock_movements(created_at);
```

# Strategy Realtime Stok

## Saat Order Dibuat
1. Sistem lock stok
2. Tambah reserved_quantity
3. Broadcast websocket

## Saat Pembayaran
1. Kurangi quantity
2. Kurangi reserved_quantity
3. Insert stock movement
4. Broadcast websocket

## Saat Order Dibatalkan
1. Release reserved_quantity
2. Broadcast websocket
